-- To find out the existing relationships between tables

SELECT

fk.name AS [FK_Constraint_Name],

OBJECT_NAME(fk.parent_object_id) AS [Child_Table],

c_parent.name AS [Child_Column],

OBJECT_NAME(fk.referenced_object_id) AS [Parent_Table],

c_referenced.name AS [Parent_Column]

FROM

sys.foreign_keys AS fk

INNER JOIN

sys.foreign_key_columns AS fkc ON fk.object_id = fkc.constraint_object_id

INNER JOIN

sys.columns AS c_parent ON fkc.parent_column_id = c_parent.column_id

AND fkc.parent_object_id = c_parent.object_id

INNER JOIN

sys.columns AS c_referenced ON fkc.referenced_column_id = c_referenced.column_id

AND fkc.referenced_object_id = c_referenced.object_id;
